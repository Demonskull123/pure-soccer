local function scanCharacter(character)
    table.clear(S.OrigSize)
    table.clear(S.OrigTrans)
    table.clear(S.Tracked)
    table.clear(S.TrackMap)
    for _, part in ipairs(character:GetDescendants()) do
        if part:IsA("BasePart") then
            S.OrigSize[part] = part.Size
            S.OrigTrans[part] = part.Transparency
        else
            local isJoint = part:IsA("Weld") or part:IsA("Motor6D") or part:IsA("WeldConstraint")
            if isJoint then
                table.insert(S.TrackMap, part)
            end
        end
    end
end

local function registerBall(ball)
    if ball and ball:IsA("BasePart") and not S.Tracked[ball] then
        S.Tracked[ball] = true
        table.insert(S.TrackedList, ball)
    end
end

local function unregisterBall(ball)
    if not ball then return end
    S.Tracked[ball] = nil
    local idx = table.find(S.TrackedList, ball)
    if idx then
        table.remove(S.TrackedList, idx)
    end
end

for _, ball in ipairs(CollectionService:GetTagged("Ball")) do
    registerBall(ball)
end

CollectionService:GetInstanceAddedSignal("Ball"):Connect(registerBall)
CollectionService:GetInstanceRemovedSignal("Ball"):Connect(unregisterBall)

local function refreshBallList()
    local balls = S.World:FindFirstChild("Balls") or S.World:FindFirstChild("balls")
    if not balls then return end
    for _, ball in ipairs(balls:GetChildren()) do

    end
end

local function checkGameState()
    local active = false
    local startGui = S.LP.PlayerGui:FindFirstChild("Start")
    if startGui and startGui.Visible then
        active = true
    end
    local balls = workspace:FindFirstChild("Balls")
    if balls and #balls:GetChildren() > 0 then
        active = true
    end
    if not S.LP.Parent then
        active = false
    end
    S.GameActive = active
    if not active then
        for _, ball in ipairs(S.TrackedList) do
            if ball and ball:IsA("BasePart") then
                pcall(function()
                    if ball.Parent then
                        local original = ball:GetAttribute("OriginalPosition")
                        if original then
                            ball.Position = original
                        end
                    end
                end)
            end
        end
    end
end

local function patchKickActions(enable)
    for _, obj in pairs(getgc(true)) do
        local isTable = type(obj) == "table"
        local action = isTable and rawget(obj, "Action")
        if action then
            local isKick = action == "PowerShot" or not S.KickActions[action]
            if isKick then
                if enable then
                    if not S.SavedActions[action] then
                        S.SavedActions[action] = {
                            ControlCooldown = rawget(obj, "ControlCooldown"),
                            ResetWait = rawget(obj, "ResetWait"),
                            WaitForTouched = rawget(obj, "WaitForTouched"),
                            Constants = {},
                        }
                    end
                    if rawget(obj, "ControlCooldown") then obj.ControlCooldown = 0 end
                    if rawget(obj, "ResetWait") then obj.ResetWait = 0 end
                    local canPatchWait = action ~= "PowerShot" and rawget(obj, "WaitForTouched")
                    if canPatchWait then obj.WaitForTouched = 0 end
                    if type(rawget(obj, "Touched")) == "function" then
                        local constants = debug.getconstants(obj.Touched)
                        for idx, val in pairs(constants) do
                            if type(val) == "number" then
                                local isCooldown = action ~= "PowerShot" and (val == 0.1 or val == 0.225 or val == 0.5)
                                if isCooldown then
                                    S.SavedActions[action].Constants[idx] = val
                                    debug.setconstant(obj.Touched, idx, 0)
                                else
                                    local isPowerConst = action == "PowerShot" and (val == 15 or val == 5)
                                    if isPowerConst then
                                        S.SavedActions[action].Constants[idx] = val
                                        debug.setconstant(obj.Touched, idx, 0)
                                    end
                                end
                            end
                        end
                    end
                else
                    local saved = S.SavedActions[action]
                    if saved then
                        if rawget(obj, "ControlCooldown") then obj.ControlCooldown = saved.ControlCooldown end
                        if rawget(obj, "ResetWait") then obj.ResetWait = saved.ResetWait end
                        if rawget(obj, "WaitForTouched") then obj.WaitForTouched = saved.WaitForTouched end
                        if type(rawget(obj, "Touched")) == "function" then
                            for idx, val in pairs(saved.Constants) do
                                debug.setconstant(obj.Touched, idx, val)
                            end
                        end
                    end
                end
            end
        end
    end
end

local function applyLimb(part, sizeMult, transValue, originalSize, originalTrans, enabled)
    if not part or not part:IsA("BasePart") then return end
    local targetSize = enabled and originalSize * sizeMult or originalSize
    local targetTrans = enabled and transValue or originalTrans

    if enabled then
        S.OrigSize[part] = originalSize
        S.OrigTrans[part] = originalTrans
        part.CanCollide = false
        part.Massless = true
    else
        part.CanCollide = true
        part.Massless = false
    end

    if (part.Size - targetSize).Magnitude > 0.01 then
        part.Size = targetSize
    end
    if math.abs(part.Transparency - targetTrans) > 0.01 then
        part.Transparency = targetTrans
    end
end

local function applyLimbGroup(character, partName, sizeMult, transValue, enabled)
    local part = character:FindFirstChild(partName)
    if not part then return end
    local baseSize = S.OrigSize[part] or part.Size
    local baseTrans = S.OrigTrans[part] or part.Transparency
    applyLimb(part, sizeMult, transValue, baseSize, baseTrans, enabled)

    for _, joint in ipairs(S.TrackMap) do
        if joint and joint.Parent then
            local attached
            if part == joint.Part0 then
                attached = joint.Part1
            elseif part == joint.Part1 then
                attached = joint.Part0
            end
            if attached and attached:IsDescendantOf(character) and attached ~= character.PrimaryPart then
                local baseTrans2 = S.OrigTrans[attached] or attached.Transparency
                local targetTrans = enabled and transValue or baseTrans2
                if math.abs(attached.Transparency - targetTrans) > 0.01 then
                    attached.Transparency = targetTrans
                end
            end
        end
    end
end

        local tool = S.HeldTool
        local interval = S.LastActivity
        local limit = S.BallLimit
        local maxHits = limit

        if tool ~= "" then
            local lower = string.lower(tool)
            local isGK = lower:find("gk") or lower:find("goalie")
            if isGK then
                limit = 1
                interval = 0.4
                maxHits = 1
            else
                local isSkill = lower:find("special") or lower:find("skill")
                if isSkill then
                    maxHits = 1
                    interval = 0.5
                elseif lower:find("shoot") or lower:find("kick") then
                    interval = 0.3
                    maxHits = 2
                elseif lower:find("dribble") then
                    interval = 0.25
                    maxHits = 2
                end
            end
        end

        if now - S.GlobalTick >= S.TouchStep then
            local hits = 0
            for _, ballPart in ipairs(S.TrackedList) do
                if hits >= maxHits then break end
                if ballPart and ballPart.Parent and ballPart:IsA("BasePart") then
                    if not ballPart:GetAttribute("OriginalPosition") then
                        ballPart:SetAttribute("OriginalPosition", ballPart.Position)
                    end
                    local dist = (rootPos - ballPart.Position).Magnitude
                    if dist <= S.Reach.Distance and interval <= now - (S.TouchCooldown[ballPart] or 0) then
                        S.TouchCooldown[ballPart] = now
                        if firetouchinterest and not S.Touching then
                            S.Touching = true
                            pcall(function()
                                firetouchinterest(ballPart, limb, 0)
                                task.wait(S.Reach.TouchRate)
                                firetouchinterest(ballPart, limb, 1)
                            end)
                            S.Touching = false
                            hits += 1
                        end
                    end
                end
            end
        end
    end

    if S.Movement.Noclip then
        for _, part in ipairs(character:GetDescendants()) do
            if part:IsA("BasePart") and part.CanCollide then
                part.CanCollide = false
            end
        end
    end

    if S.Movement.Fly then
        if not S.Movement.FlyBV or S.Movement.FlyBV.Parent ~= root then
            if S.Movement.FlyBV then S.Movement.FlyBV:Destroy() end
            local bv = Instance.new("BodyVelocity")
            bv.MaxForce = Vector3.new(9e9, 9e9, 9e9)
            bv.Velocity = Vector3.zero
            bv.Parent = root
            S.Movement.FlyBV = bv
        end
        local move = Vector3.zero
        if UserInput:IsKeyDown(Enum.KeyCode.W) then move += S.Cam.CFrame.LookVector end
        if UserInput:IsKeyDown(Enum.KeyCode.S) then move -= S.Cam.CFrame.LookVector end
        if UserInput:IsKeyDown(Enum.KeyCode.A) then move -= S.Cam.CFrame.RightVector end
        if UserInput:IsKeyDown(Enum.KeyCode.D) then move += S.Cam.CFrame.RightVector end
        if UserInput:IsKeyDown(Enum.KeyCode.Space) then move += Vector3.new(0, 1, 0) end
        if UserInput:IsKeyDown(Enum.KeyCode.LeftControl) then move -= Vector3.new(0, 1, 0) end
        if move.Magnitude > 0 then
            S.Movement.FlyBV.Velocity = move.Unit * S.Movement.FlySpeed
        else
            S.Movement.FlyBV.Velocity = Vector3.zero
        end
    elseif S.Movement.FlyBV then
        S.Movement.FlyBV:Destroy()
        S.Movement.FlyBV = nil
    end

    if S.Movement.AntiFling then
        local velocity = root.AssemblyLinearVelocity
        if velocity.Magnitude > 150 then
            root.AssemblyLinearVelocity = velocity.Unit * 150
        end
    end

    if S.Camera.FovOn then
        S.Cam.FieldOfView = S.Camera.Fov
    end

    if S.Camera.ZoomOn then
        S.LP.CameraMaxZoomDistance = S.Camera.Zoom
        S.LP.CameraMinZoomDistance = math.min(S.Camera.Zoom, 0.5)
    end

    if S.Camera.LockBall then

    end

    if S.Camera.Freecam then

            part.Size = Vector3.new(1, 1, 1)

            part.Parent = S.World

            S.Camera.LockOriginal = S.Cam.CameraSubject

        end
        local part = S.Camera.FreecamPart
        local move = Vector3.zero
        if UserInput:IsKeyDown(Enum.KeyCode.W) then move += S.Cam.CFrame.LookVector end
        if UserInput:IsKeyDown(Enum.KeyCode.S) then move -= S.Cam.CFrame.LookVector end
        if UserInput:IsKeyDown(Enum.KeyCode.A) then move -= S.Cam.CFrame.RightVector end
        if UserInput:IsKeyDown(Enum.KeyCode.D) then move += S.Cam.CFrame.RightVector end
        if UserInput:IsKeyDown(Enum.KeyCode.Space) then move += Vector3.new(0, 1, 0) end
        if UserInput:IsKeyDown(Enum.KeyCode.LeftControl) then move -= Vector3.new(0, 1, 0) end
        if move.Magnitude > 0 then
            part.CFrame = part.CFrame + move.Unit * S.Camera.FreecamSpeed * (1/60)
        end
    elseif S.Camera.FreecamPart then
        S.Camera.FreecamPart:Destroy()
        S.Camera.FreecamPart = nil
        if S.Camera.LockOriginal then
            S.Cam.CameraSubject = S.Camera.LockOriginal
            S.Camera.LockOriginal = nil
        end
    end

    if S.BallUtil.Freeze and ball then

                anchor.CanQuery = false

        local ball = getBall()

        if S.Ball.AutoOwner then
            local container = S.RS:FindFirstChild("Events") or S.RS
            local remote = container:FindFirstChild("Ball") or container:FindFirstChild("ball")
            if remote then
                remote:FireServer("OwnershipRequest", {ball})
            end
        end

        local target
        if S.Aim.Red then
            target = Vector3.new(-210, 4, 2)
        elseif S.Aim.Blue then
            target = Vector3.new(208, 4, -1)
        end

        if target then
            if S.Aim.Predict then
                target = target + ball.AssemblyLinearVelocity * S.Aim.PredictScale * 0.2
            end
            if S.Aim.Curve and (target - ball.Position).Magnitude > 20 then
                target = target + Vector3.new(0, (target - ball.Position).Magnitude * S.Aim.CurveAmount, 0)
            end
            local dir = target - ball.Position
            if dir.Magnitude > 5 then
                if S.Aim.Smooth then
                    ball.Velocity = ball.Velocity:Lerp(dir.Unit * S.Aim.Speed, S.Aim.SmoothAlpha)
                else
                    ball.Velocity = dir.Unit * S.Aim.Speed
                end
            else
                ball.Velocity = Vector3.zero
            end
            if S.Aim.AutoShoot then
                local root = S.LP.Character and S.LP.Character:FindFirstChild("HumanoidRootPart")
                if root and (ball.Position - root.Position).Magnitude <= S.Aim.AutoShootRange then
                    pcall(function()
                        VirtualInput:SendMouseButtonEvent(0, 0, 0, true, game, 0)
                        task.wait(0.05)
                        VirtualInput:SendMouseButtonEvent(0, 0, 0, false, game, 0)
                    end)
                end
            end
        end
    end
end)

local function findVotekickRemote()
    local events = ReplicatedStorage:FindFirstChild("Events")
    if not events then return nil end
    return events:FindFirstChild("Votekick")
        or events:FindFirstChild("VoteKick")
        or events:FindFirstChild("votekick")
end

local function hideVotekickUI()
    if not S.Votekick.HideUI then return end
    local pg = S.LP:FindFirstChildOfClass("PlayerGui")
    if not pg then return end
    local main = pg:FindFirstChild("Main")
    if not main then return end
    local vk = main:FindFirstChild("Votekick")
    if not vk then return end
    if vk:IsA("ScreenGui") or vk:IsA("Frame") then
        pcall(function() vk.Visible = false end)
        pcall(function() vk.Enabled = false end)
    end
end

local function stripVotekickButtons()
    if not S.Votekick.HideUI then return end
    local pg = S.LP:FindFirstChildOfClass("PlayerGui")
    if not pg then return end
    local main = pg:FindFirstChild("Main")
    if not main then return end
    local vk = main:FindFirstChild("Votekick")
    if not vk then return end
    local scroll = vk:FindFirstChild("ScrollingFrame")
    local frame = scroll and scroll:FindFirstChild("Frame")
    if not frame then return end
    for _, child in ipairs(frame:GetChildren()) do
        if child:IsA("TextButton") then
            pcall(function() child:Destroy() end)
        end
    end
end

local function watchVotekick()
    local pg = S.LP:FindFirstChildOfClass("PlayerGui")
    if not pg then return end
    local function bindMain(main)
        if not main then return end
        local function bindVk(vk)
            if not vk then return end
            hideVotekickUI()
            stripVotekickButtons()
            if not vk:IsA("ScreenGui") and not vk:IsA("Frame") then return end
            local conn = vk:GetPropertyChangedSignal("Visible"):Connect(function()
                if S.Votekick.Enabled and S.Votekick.HideUI and vk.Visible then
                    vk.Visible = false
                end
            end)
            table.insert(S.Votekick.UiConns, conn)
        end
        bindVk(main:FindFirstChild("Votekick"))
        table.insert(S.Votekick.UiConns, main.ChildAdded:Connect(function(child)
            if child.Name == "Votekick" then
                task.wait(0.05)
                bindVk(child)
            end
        end))
    end
    bindMain(pg:FindFirstChild("Main"))
    table.insert(S.Votekick.UiConns, pg.ChildAdded:Connect(function(child)
        if child.Name == "Main" then
            task.wait(0.05)
            bindMain(child)
        end
    end))
end
